        .txtLogradouro = ""

        .txtFoto = ""

End Sub

Private Sub cmdFoto_Click()

    Dim fd As FileDialog

    Set fd = Application.FileDialog(msoFileDialogFilePicker)

    With fd

        .Title = "Selecione uma foto"

        .Filters.Clear
        .Filters.Add "Imagens", "*.jpg;*.jpeg;*.png;*.bmp"

        If .Show = -1 Then

            txtFoto.Value = .SelectedItems(1)

            frmAluno.imgFoto.Picture = LoadPicture(txtFoto.Value)

        End If

    End With

End Sub

Private Sub cmdSalvar_Click()
    If lngLinha = 0 Then
        lngLinha = wsCadAluno.Cells(wsCadAluno.Rows.Count, 1).End(xlUp).Row
        'If lngLinha < 2 Then
            'lngLinha = lngLinha + 1
        'End If
        lngLinha = lngLinha + 1
    End If
    CadastrarAluno lngLinha
    MsgBox "Aluno(a): " & wsCadAluno.Cells(lngLinha, 2).Value & vbCrLf & _
        "Código: " & wsCadAluno.Cells(lngLinha, 1).Value, _
        vbInformation + vbOKOnly, _
        "Sucesso"
    LimparFormulario
    lngLinha = 0
    CarregarListaAlunos
End Sub
